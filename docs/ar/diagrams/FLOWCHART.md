## 1. تدفق حجز الخدمة

```mermaid
flowchart TD
    A["المستخدم يتصفح الخدمات"] --> B["اختيار الفرع/الفني/الوقت"]
    B --> C["كتابة ملاحظة"]
    C --> D{"اختيار قسيمة؟"}
    D -->|"استخدام"| E["خصم بقيمة القسيمة"]
    D -->|"عدم الاستخدام"| F["الطلب بالسعر الأصلي"]
    E --> G["حساب سعر الطلب (بدون استهلاك)<br/>حساب نقي عبر PriceCalculator<br/>القسيمة fixed/percent + بطاقة المرات times<br/>min_amount بناءً على السعر الأصلي"]
    F --> G
    G --> H["قراءة اتفاقية الخدمة"]
    H --> I["تقديم الطلب"]
    I --> J{"قفل الفني عبر Redis<br/>SETNX 3 دقائق"}
    J -->|"نجح القفل"| K["إنشاء طلب pending"]
    J -->|"مقفل بالفعل"| L["إشعار بأن الفني مشغول"]
    K --> M{"المبلغ المستحق؟"}
    M -->|"صفر"| N["عبور مباشر FREE<br/>transaction_id = 'FREE'+رقم دفعة الدفع<br/>الطلب → paid"]
    M -->|"الدفع بالرصيد"| B1["خصم رصيد المحفظة<br/>تسجيل إدخال wallet_txn<br/>الطلب → paid"]
    M -->|"المبلغ > 0"| O{"طريقة الدفع"}
    O -->|"WeChat"| OW["استدعاء الدفع عبر WeChat<br/>pay_lock لمنع الدفع المكرر المتزامن"]
    O -->|"الرصيد"| B1
    OW --> P{"نتيجة الدفع"}
    B1 --> S
    P -->|"نجاح"| Q["استهلاك رد نداء نجاح الدفع<br/>markOrderPaid نقطة استهلاك واحدة<br/>خصم ذري للقسيمة/بطاقة المرات<br/>الطلب → paid"]
    P -->|"فشل/إلغاء"| R["يبقى الطلب pending<br/>إلغاء تلقائي بعد 15 دقيقة"]
    N --> S["الفني يؤكد بدء الخدمة"]
    Q --> S
    S --> T["الطلب → serving"]
    T --> U["اكتمال الخدمة"]
    U --> V["إلغاء تحقق الفني بالمسح الضوئي"]
    V --> W["الطلب → completed"]
    W --> X["تقييم المستخدم (نص + صورة)"]
    X --> Y["الطلب → reviewed ✅"]

    style A fill:#e3f2fd,stroke:#1565c0,color:#333
    style Y fill:#c8e6c9,stroke:#2e7d32,color:#333
    style L fill:#ffcdd2,stroke:#c62828,color:#333
    style R fill:#fff9c4,stroke:#f9a825,color:#333
    style N fill:#c8e6c9,stroke:#2e7d32,color:#333
```

## 2. تدفق الدفع والاسترداد

```mermaid
flowchart TD
    subgraph 支付流程["تدفق الدفع المباشر"]
        P1["إنشاء سجل الدفع"] --> P2["طلب موحد عبر WeChat<br/>pay_lock لمنع التزامن<br/>out_trade_no = order_no تكراري الأثر"]
        P2 --> P3["الواجهة الأمامية تستدعي الدفع<br/>اختيار طريقة الدفع"]
        P3 -->|"الرصيد"| PB["خصم رصيد المحفظة<br/>تسجيل إدخال wallet_txn<br/>تكراري الأثر، خصم مرة واحدة فقط"]
        P3 -->|"WeChat"| P4["رد نداء WeChat notify"]
        P4 --> P5["نجاح التحقق من التوقيع"]
        PB --> P6["markOrderPaid تكراري الأثر<br/>القسيمة/بطاقة المرات تُستهلك مرة واحدة فقط"]
        P5 --> P6
        P6 --> P7["الطلب → paid<br/>إشعار المستخدم + الفني"]
    end

    subgraph 退款流程["تدفق الاسترداد"]
        R1["المستخدم يطلب الاسترداد<br/>refund_lock لمنع التزامن"] --> R2{"تحديد قاعدة الاسترداد"}
        R2 -->|"الطلب خلال ≤15 دقيقة أو البعد عن البداية >6 ساعات"| R3["استرداد 100%"]
        R2 -->|"البعد عن البداية ≤6 ساعات"| R4["استرداد 90%"]
        R2 -->|"بدأ ولم يُؤكد"| R5["استرداد 80%"]
        R2 -->|"بعد تأكيد الخدمة"| R6["لا استرداد"]
        R3 --> R7["الطلب → refunding"]
        R4 --> R7
        R5 --> R7
        R7 --> R8["موافقة من مستويين<br/>مدير الفرع → المالية"]
        R8 --> R9["استرداد من مرحلتين<br/>إنشاء سجل الاسترداد داخل المعاملة<br/>إدخال/إخراج استرداد WeChat خارج المعاملة"]
        R9 -->|"فشل WeChat"| R10["إرجاع الطلب إلى PAID<br/>إمكانية إعادة محاولة الاسترداد"]
        R9 -->|"نجاح الاسترداد"| R11["الطلب → refunded<br/>إرجاع WeChat عبر نفس المسار / إعادة شحن الرصيد<br/>إرجاع القسيمة + استرداد النقاط"]
    end

    style P6 fill:#c8e6c9,stroke:#2e7d32,color:#333
    style R6 fill:#ffcdd2,stroke:#c62828,color:#333
    style R11 fill:#c8e6c9,stroke:#2e7d32,color:#333
    style R10 fill:#fff9c4,stroke:#f9a825,color:#333
```
