## 1. সার্ভিস বুকিং ফ্লো

```mermaid
flowchart TD
    A["ইউজার সার্ভিস আইটেম ব্রাউজ করে"] --> B["শাখা/টেকনিশিয়ান/সময় নির্বাচন"]
    B --> C["নোট লিখুন"]
    C --> D{"কুপন নির্বাচন করবেন?"}
    D -->|"ব্যবহার"| E["কুপন দিয়ে মূল্য ছাড়"]
    D -->|"ব্যবহার করবেন না"| F["মূল দামে অর্ডার"]
    E --> G["অর্ডার মূল্য গণনা (ব্যবহার হয় না)<br/>PriceCalculator বিশুদ্ধ হিসাব<br/>কুপন fixed/percent + টাইমস কার্ড times<br/>min_amount মূল দামের ভিত্তিতে"]
    F --> G
    G --> H["সার্ভিস চুক্তি পড়ুন"]
    H --> I["অর্ডার সাবমিট"]
    I --> J{"Redis টেকনিশিয়ান লক<br/>SETNX 3 মিনিট"}
    J -->|"লক সফল"| K["অর্ডার তৈরি pending"]
    J -->|"আগে লক করা"| L["টেকনিশিয়ান ব্যস্ত বার্তা"]
    K --> M{"পরিশোধযোগ্য অ্যামাউন্ট?"}
    M -->|"শূন্য টাকা"| N["FREE সরাসরি<br/>transaction_id = 'FREE'+পেমেন্ট নম্বর<br/>অর্ডার → paid"]
    M -->|"ব্যালেন্স পেমেন্ট"| B1["ওয়ালেট ব্যালেন্স কাটা<br/>wallet_txn এন্ট্রি<br/>অর্ডার → paid"]
    M -->|"অ্যামাউন্ট > 0"| O{"পেমেন্ট পদ্ধতি"}
    O -->|"WeChat"| OW["WeChat পেমেন্ট কল<br/>pay_lock কনকারেন্ট রিপিট পেমেন্ট প্রতিরোধ"]
    O -->|"ব্যালেন্স"| B1
    OW --> P{"পেমেন্ট ফলাফল"}
    B1 --> S
    P -->|"সফল"| Q["পেমেন্ট সফল কলব্যাক কনজিউম<br/>markOrderPaid একক কনজিউম পয়েন্ট<br/>কুপন/টাইমস কার্ড অ্যাটমিক কাটা<br/>অর্ডার → paid"]
    P -->|"ব্যর্থ/বাতিল"| R["অর্ডার pending থাকে<br/>১৫ মিনিট পর স্বয়ংক্রিয় বাতিল"]
    N --> S["টেকনিশিয়ান সার্ভিস শুরু নিশ্চিত করে"]
    Q --> S
    S --> T["অর্ডার → serving"]
    T --> U["সার্ভিস সম্পন্ন"]
    U --> V["টেকনিশিয়ান স্ক্যান-ভেরিফিকেশন"]
    V --> W["অর্ডার → completed"]
    W --> X["ইউজার রিভিউ (টেক্সট + ছবি)"]
    X --> Y["অর্ডার → reviewed ✅"]

    style A fill:#e3f2fd,stroke:#1565c0,color:#333
    style Y fill:#c8e6c9,stroke:#2e7d32,color:#333
    style L fill:#ffcdd2,stroke:#c62828,color:#333
    style R fill:#fff9c4,stroke:#f9a825,color:#333
    style N fill:#c8e6c9,stroke:#2e7d32,color:#333
```

## 2. পেমেন্ট ও রিফান্ড ফ্লো

```mermaid
flowchart TD
    subgraph 支付流程["ফরওয়ার্ড পেমেন্ট ফ্লো"]
        P1["পেমেন্ট রেকর্ড তৈরি"] --> P2["WeChat ইউনিফাইড অর্ডার<br/>pay_lock কনকারেন্সি প্রতিরোধ<br/>out_trade_no = order_no আইডেমপোটেন্ট"]
        P2 --> P3["ফ্রন্টএন্ড পেমেন্ট শুরু<br/>পেমেন্ট পদ্ধতি নির্বাচন"]
        P3 -->|"ব্যালেন্স"| PB["ওয়ালেট ব্যালেন্স কাটা<br/>wallet_txn এন্ট্রি<br/>আইডেমপোটেন্ট শুধু একবার কাটে"]
        P3 -->|"WeChat"| P4["WeChat কলব্যাক notify"]
        P4 --> P5["সাইন ভেরিফিকেশন সফল"]
        PB --> P6["markOrderPaid আইডেমপোটেন্ট<br/>কুপন/টাইমস কার্ড শুধু এখানেই কনজিউম হয়"]
        P5 --> P6
        P6 --> P7["অর্ডার → paid<br/>ইউজার + টেকনিশিয়ানকে নোটিফিকেশন"]
    end

    subgraph 退款流程["রিফান্ড ফ্লো"]
        R1["ইউজার রিফান্ড আবেদন<br/>refund_lock কনকারেন্সি প্রতিরোধ"] --> R2{"রিফান্ড নিয়ম নির্ধারণ"}
        R2 -->|"অর্ডার ≤১৫ মিনিট বা শুরু >৬ ঘণ্টা আগে"| R3["রিফান্ড 100%"]
        R2 -->|"শুরু ≤৬ ঘণ্টা আগে"| R4["রিফান্ড 90%"]
        R2 -->|"শুরু হয়েছে নিশ্চিত হয়নি"| R5["রিফান্ড 80%"]
        R2 -->|"সার্ভিস নিশ্চিত হওয়ার পর"| R6["রিফান্ড নেই"]
        R3 --> R7["অর্ডার → refunding"]
        R4 --> R7
        R5 --> R7
        R7 --> R8["দ্বি-স্তর অনুমোদন<br/>শাখা ম্যানেজার → ফাইন্যান্স"]
        R8 --> R9["দ্বি-পর্যায় রিফান্ড<br/>ট্রানজেকশনের ভেতরে রিফান্ড রেকর্ড তৈরি<br/>ট্রানজেকশনের বাইরে WeChat রিফান্ড IO"]
        R9 -->|"WeChat ব্যর্থ"| R10["অর্ডার PAID-এ রোলব্যাক<br/>রিফান্ড পুনরায় চেষ্টাযোগ্য"]
        R9 -->|"রিফান্ড সফল"| R11["অর্ডার → refunded<br/>WeChat মূল পথে ফেরত / ব্যালেন্স রিফিল<br/>কুপন ফেরত + পয়েন্ট রিবেট"]
    end

    style P6 fill:#c8e6c9,stroke:#2e7d32,color:#333
    style R6 fill:#ffcdd2,stroke:#c62828,color:#333
    style R11 fill:#c8e6c9,stroke:#2e7d32,color:#333
    style R10 fill:#fff9c4,stroke:#f9a825,color:#333
```
